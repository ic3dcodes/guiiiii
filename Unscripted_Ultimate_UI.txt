local function create(className, props, parent)
    local instance = Instance.new(className)
    for key, value in pairs(props or {}) do instance[key] = value end
    instance.Parent = parent
    return instance
end

local function corner(parent, radius)
    return create("UICorner", {CornerRadius = UDim.new(0, radius or 10)}, parent)
end

local function stroke(parent, color, thickness, transparency)
    return create("UIStroke", {
        Color = color or Theme.border,
        Thickness = thickness or 1,
        Transparency = transparency == nil and 0.18 or transparency,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
    }, parent)
end

local function padding(parent, left, right, top, bottom)
    return create("UIPadding", {
        PaddingLeft = UDim.new(0, left or 0),
        PaddingRight = UDim.new(0, right or 0),
        PaddingTop = UDim.new(0, top or 0),
        PaddingBottom = UDim.new(0, bottom or 0),
    }, parent)
end

local function animate(object, duration, properties, style, direction)
    if not object or not object.Parent then return nil end
    local info = TweenInfo.new(
        duration or 0.16,
        style or Enum.EasingStyle.Quint,
        direction or Enum.EasingDirection.Out
    )
    local tw = TweenService:Create(object, info, properties)
    tw:Play()
    return tw
end

local function setPage(name)
    if not pages[name] then return end
    state.page = name
    for pageName, page in pairs(pages) do
        page.Visible = pageName == name
    end
    for pageName, button in pairs(navButtons) do
        local active = pageName == name
        animate(button, state.reduceMotion and 0.04 or 0.16, {
            BackgroundColor3 = active and Theme.accent or Theme.card,
            TextColor3 = active and Color3.new(1, 1, 1) or Theme.muted,
        })
        local edge = button:FindFirstChildOfClass("UIStroke")
        if edge then animate(edge, state.reduceMotion and 0.04 or 0.16, {Color = active and Theme.accent or Theme.border}) end
    end
    local activePage = pages[name]
    if activePage and not state.reduceMotion then
        activePage.Position = UDim2.new(0, 5, 0, 0)
        animate(activePage, 0.18, {Position = UDim2.new(0, 0, 0, 0)})
    end
end

local function bindAccent(callback)
    table.insert(accentBindings, callback)
    callback(Theme.accent)
end

local function applyAccent(color)
    Theme.accent = color
    for i = #accentBindings, 1, -1 do
        local ok, keep = pcall(accentBindings[i], color)
        if not ok or keep == false then table.remove(accentBindings, i) end
    end
end

local function toast(message)
    if not state.notifications or state.closed then return end
    local box = create("Frame", {
        AnchorPoint = Vector2.new(1, 0),
        Position = UDim2.new(1, -12, 0, 64),
        Size = UDim2.new(0, 220, 0, 37),
        BackgroundColor3 = Theme.card,
        BackgroundTransparency = 0.06,
        BorderSizePixel = 0,
        ZIndex = 110,
    }, toastLayer)
    corner(box, 10)
    local edge = stroke(box, Theme.accent, 1, 0.05)
    bindAccent(function(color)
        if not edge.Parent then return false end
        edge.Color = color
    end)
    local label = create("TextLabel", {
        Position = UDim2.fromOffset(11, 0),
        Size = UDim2.new(1, -22, 1, 0),
        BackgroundTransparency = 1,
        Text = message,
        TextColor3 = Theme.text,
        Font = Enum.Font.GothamMedium,
        TextSize = 10,
        TextXAlignment = Enum.TextXAlignment.Left,
        TextTruncate = Enum.TextTruncate.AtEnd,
        TextTransparency = 1,
        ZIndex = 111,
    }, box)
    local from = box.Position
    box.Position = from + UDim2.fromOffset(0, -8)
    animate(box, 0.2, {Position = from})
    animate(label, 0.2, {TextTransparency = 0})
    task.delay(1.65, function()
        if box.Parent then
            animate(box, 0.16, {Position = from + UDim2.fromOffset(0, -6), BackgroundTransparency = 1})
            animate(label, 0.13, {TextTransparency = 1})
            task.delay(0.2, function() if box.Parent then box:Destroy() end end)
        end
    end)
end

local function particleBurst(absolutePosition, color)
    if not state.particles or state.reduceMotion or not toastLayer.Parent then return end
    if state.particleCount > 85 then return end
    local center = absolutePosition - toastLayer.AbsolutePosition
    for i = 1, 6 do
        state.particleCount += 1
        local dot = create("Frame", {
            AnchorPoint = Vector2.new(0.5, 0.5),
            Position = UDim2.fromOffset(center.X, center.Y),
            Size = UDim2.fromOffset(math.random(3, 6), math.random(3, 6)),
            BackgroundColor3 = color or Theme.accent,
            BorderSizePixel = 0,
            ZIndex = 105,
        }, toastLayer)
        corner(dot, 100)
        local angle = math.rad((360 / 6) * i + math.random(-16, 16))
        local distance = math.random(18, 38)
        animate(dot, 0.38, {
            Position = UDim2.fromOffset(center.X + math.cos(angle) * distance, center.Y + math.sin(angle) * distance),
            Size = UDim2.fromOffset(0, 0),
            BackgroundTransparency = 1,
        }, Enum.EasingStyle.Quint)
        task.delay(0.42, function()
            if dot.Parent then dot:Destroy() end
            state.particleCount = math.max(0, state.particleCount - 1)
        end)
    end
end

local function makePage(name, icon, order, description)
    local page = create("ScrollingFrame", {
        Name = name .. "Page",
        Size = UDim2.fromScale(1, 1),
        BackgroundColor3 = Theme.panel,
        BorderSizePixel = 0,
        ScrollBarThickness = 3,

        CanvasSize = UDim2.new(0, 0, 0, 0),
        AutomaticCanvasSize = Enum.AutomaticSize.Y,
        ScrollingDirection = Enum.ScrollingDirection.Y,
        Visible = false,
        ZIndex = 3,
    }, pageHolder)
    corner(page, 12)
    stroke(page, Theme.border, 1, 0.48)
    padding(page, 12, 12, 11, 15)
    create("UIListLayout", {
        Padding = UDim.new(0, 8),
        SortOrder = Enum.SortOrder.LayoutOrder,
    }, page)
    pages[name] = page

    local nav = create("TextButton", {
        Name = name .. "Nav",
        LayoutOrder = order,
        Size = UDim2.fromOffset(37, 39),
        BackgroundColor3 = Theme.card,
        Text = icon,
        TextColor3 = Theme.muted,
        Font = Enum.Font.GothamBold,
        TextSize = 18,
        AutoButtonColor = false,
        BorderSizePixel = 0,
        ZIndex = 5,
    }, sidebar)
    corner(nav, 10)
    local navEdge = stroke(nav, Theme.border, 1, 0.5)
    navButtons[name] = nav
    bindAccent(function(color)
        if nav.Parent then
            navEdge.Color = name == state.page and color or Theme.border
            nav.BackgroundColor3 = name == state.page and color or Theme.card
            nav.TextColor3 = name == state.page and Color3.new(1,1,1) or Theme.muted
        end
    end)
    nav.Activated:Connect(function()
        if state.page == name then return end
        setPage(name)
        particleBurst(nav.AbsolutePosition + nav.AbsoluteSize / 2, Theme.accent)
    end)
    return page
end

local function addToggle(parent, title, description, default, key)
    local h = description and 57 or 43
    local row = card(parent, h)
    label(row, title, UDim2.fromOffset(12, description and 6 or 0), UDim2.new(1, -72, 0, description and 20 or h), {textSize = 11})
    if description then
        label(row, description, UDim2.fromOffset(12, 28), UDim2.new(1, -24, 0, 17), {color = Theme.muted, textSize = 9, truncate = true})
    end
    local track = create("TextButton", {
        AnchorPoint = Vector2.new(1, 0.5),
        Position = UDim2.new(1, -11, 0.5, 0),
        Size = UDim2.fromOffset(38, 21),
        BackgroundColor3 = default and Theme.accent or Theme.off,
        Text = "",
        AutoButtonColor = false,
        BorderSizePixel = 0,
        ZIndex = 6,
    }, row)
    corner(track, 20)
    local knob = create("Frame", {
        AnchorPoint = Vector2.new(0, 0.5),
        Position = default and UDim2.new(1, -18, 0.5, 0) or UDim2.new(0, 3, 0.5, 0),
        Size = UDim2.fromOffset(14, 14),
        BackgroundColor3 = Color3.new(1,1,1),
        BorderSizePixel = 0,
        ZIndex = 7,
    }, track)
    corner(knob, 20)
    local on = default == true
    values[key or title] = on
    bindAccent(function(color)
        if track.Parent and on then track.BackgroundColor3 = color end
    end)
    local function set(value, notify)
        on = value == true
        values[key or title] = on
        animate(track, 0.14, {BackgroundColor3 = on and Theme.accent or Theme.off})
        animate(knob, 0.16, {Position = on and UDim2.new(1, -18, 0.5, 0) or UDim2.new(0, 3, 0.5, 0)})
        if notify then
            particleBurst(track.AbsolutePosition + track.AbsoluteSize / 2, Theme.accent)
            if notify ~= "silent" then toast(title .. (on and " enabled" or " disabled") .. " · UI only") end
        end
    end
    track.Activated:Connect(function() set(not on, true) end)
    return {row = row, set = set, get = function() return on end}
end

local function addDropdown(parent, title, options, default, key)
    local row = card(parent, 57)
    label(row, title, UDim2.fromOffset(12, 3), UDim2.new(1, -24, 0, 17), {textSize = 10})
    local pick = create("TextButton", {
        Position = UDim2.new(0, 9, 0, 24),
        Size = UDim2.new(1, -18, 0, 25),
        BackgroundColor3 = Theme.secondary,
        Text = "  " .. default .. "     ▾",
        TextColor3 = Theme.text,
        Font = Enum.Font.GothamMedium,
        TextSize = 10,
        TextXAlignment = Enum.TextXAlignment.Left,
        AutoButtonColor = false,
        BorderSizePixel = 0,
        ZIndex = 6,
    }, row)
    corner(pick, 7)
    local edge = stroke(pick, Theme.border, 1, 0.5)
    bindAccent(function(color) if edge.Parent then edge.Color = color end end)
    local idx = table.find(options, default) or 1
    values[key or title] = default
    local scale = create("UIScale", {Scale = 1}, pick)
    pick.Activated:Connect(function()
        idx = idx % #options + 1
        values[key or title] = options[idx]
        pick.Text = "  " .. options[idx] .. "     ▾"
        if not state.reduceMotion then
            scale.Scale = 0.97
            animate(scale, 0.13, {Scale = 1}, Enum.EasingStyle.Back)
        end
        particleBurst(pick.AbsolutePosition + pick.AbsoluteSize / 2, Theme.accent)
        toast(title .. ": " .. options[idx] .. " · placeholder")
    end)
    return row
end

-- Wire behavior directly to the toggle API so state never depends on searching descendants.
local function syncToggles()
    state.particles = values.Particles ~= false
    state.reduceMotion = values.ReduceMotion == true
    state.notifications = values.Notifications ~= false
    state.rgb = values.RGB == true
end
-- Add a second activation listener to update the local effect system after the toggle state changes.
for _, toggleData in ipairs({particleToggle, motionToggle, notifyToggle, rgbToggle}) do
    local button = toggleData.row:FindFirstChildWhichIsA("TextButton")
    if button then
        button.Activated:Connect(function()
            task.defer(function()
                syncToggles()
                if not state.notifications then return end
                if toggleData == particleToggle then toast("Particle bursts " .. (state.particles and "on" or "off"))
                elseif toggleData == motionToggle then toast("Reduced motion " .. (state.reduceMotion and "on" or "off"))
                elseif toggleData == notifyToggle then
                    if state.notifications then toast("Notifications on") end
                elseif toggleData == rgbToggle then toast("RGB " .. (state.rgb and "enabled" or "disabled")) end
            end)
        end)
    end
end

-- Responsive sizing uses the live viewport; portrait phones get a tall, narrow panel.
local function updateResponsiveSize()
    camera = workspace.CurrentCamera
    if not camera then return end
    local viewport = camera.ViewportSize
    local width, height
    if viewport.X < 700 then
        width = math.max(290, math.min(430, viewport.X * 0.92))
        height = math.max(300, math.min(660, viewport.Y * 0.80))
    else
        width = math.max(520, math.min(780, viewport.X * 0.76))
        height = math.max(420, math.min(620, viewport.Y * 0.78))
    end
    width = math.min(width, math.max(270, viewport.X - 16))
    height = math.min(height, math.max(280, viewport.Y - 50))
    main.Size = UDim2.fromOffset(width, height)
    if not state.minimized then state.windowSize = main.Size end
end
updateResponsiveSize()
if camera then camera:GetPropertyChangedSignal("ViewportSize"):Connect(updateResponsiveSize) end
workspace:GetPropertyChangedSignal("CurrentCamera"):Connect(function()
    camera = workspace.CurrentCamera
    if camera then camera:GetPropertyChangedSignal("ViewportSize"):Connect(updateResponsiveSize); updateResponsiveSize() end
end)

-- Drag support for mouse and touch.
local function makeDraggable(handle, target)
    local dragging = false
    local dragStart, startPosition, dragInput
    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPosition = target.Position
            dragInput = input
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    handle.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then dragInput = input end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input == dragInput or input.UserInputType == Enum.UserInputType.Touch) and dragStart and startPosition then
            local delta = input.Position - dragStart
            target.Position = UDim2.new(
                startPosition.X.Scale, startPosition.X.Offset + delta.X,
                startPosition.Y.Scale, startPosition.Y.Offset + delta.Y
            )
        end
    end)
end
makeDraggable(header, main)
makeDraggable(orb, orb)

    end)
end

-- Subtle looping rotation / lighting effects. RGB only updates registered accent channels.
local rgbAccumulator = 0
local orbitAngle = 0
local heartbeatConnection
heartbeatConnection = RunService.Heartbeat:Connect(function(dt)
    if state.closed or not gui.Parent then
        if heartbeatConnection then heartbeatConnection:Disconnect() end
        return
    end
    if not state.minimized and main.Parent then
        if not state.reduceMotion then
            orbitAngle = (orbitAngle + dt * 24) % 360
            orbit.Rotation = orbitAngle
            emblemGradient.Rotation = (emblemGradient.Rotation + dt * 18) % 360
            orbGradient.Rotation = (orbGradient.Rotation + dt * 15) % 360
        end
    end
    if state.rgb then
        rgbAccumulator += dt
        if rgbAccumulator >= 0.075 then
            local step = math.min(rgbAccumulator, 0.15)
            rgbAccumulator = 0
            state.hue = (state.hue + step * 0.105) % 1
            applyAccent(Color3.fromHSV(state.hue, 0.72, 1))
        end
    end
end)
